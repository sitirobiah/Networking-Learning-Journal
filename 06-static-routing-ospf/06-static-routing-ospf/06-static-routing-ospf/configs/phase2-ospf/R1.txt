# Phase 2 - OSPF area 0 - R1 (HQ)
# Step 1: remove the static route to the Branch LAN (keep the default route)
delete routing-options static route 10.10.3.0/24
commit

# Step 2: enable OSPF (em2.0 to the Internet host stays out of OSPF)
set routing-options router-id 10.255.255.1
set protocols ospf area 0.0.0.0 interface em0.0
set protocols ospf area 0.0.0.0 interface em1.0
set protocols ospf area 0.0.0.0 interface em3.0 passive
set protocols ospf area 0.0.0.0 interface lo0.0 passive
commit

# Step 4: advertise the default route into OSPF
set policy-options policy-statement EXPORT-DEFAULT term 1 from protocol static
set policy-options policy-statement EXPORT-DEFAULT term 1 from route-filter 0.0.0.0/0 exact
set policy-options policy-statement EXPORT-DEFAULT term 1 then accept
set protocols ospf export EXPORT-DEFAULT
commit
