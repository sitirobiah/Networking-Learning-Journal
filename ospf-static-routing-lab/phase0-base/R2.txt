# Phase 0 - Base config - R2 (WAN)
configure
set system host-name R2
# Root password is set interactively (type the password twice when asked):
set system root-authentication plain-text-password
set interfaces em0 unit 0 family inet address 10.0.12.2/30
set interfaces em1 unit 0 family inet address 10.0.23.1/30
set interfaces lo0 unit 0 family inet address 10.255.255.2/32
commit
